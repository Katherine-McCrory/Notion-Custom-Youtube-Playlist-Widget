document.getElementById('copy-notion-embed-btn').addEventListener('click', () => { const generatedCode = document.getElementById('generated-notion-url').textContent.trim(); const btn = document.getElementById('copy-notion-embed-btn'); function showSuccess() { const originalHTML = btn.innerHTML; btn.innerHTML = ` Copied to Clipboard!`; btn.classList.replace('bg-violet-600', 'bg-emerald-600'); setTimeout(() => { btn.innerHTML = originalHTML; btn.classList.replace('bg-emerald-600', 'bg-violet-600'); }, 2500); } // 1. Try modern navigator.clipboard API if (navigator.clipboard && window.isSecureContext) { navigator.clipboard.writeText(generatedCode).then(showSuccess).catch(() => { // 2. Fallback if iframe permissions block navigator.clipboard fallbackCopyText(generatedCode, showSuccess); }); } else { // 3. Fallback for contexts without navigator.clipboard fallbackCopyText(generatedCode, showSuccess); } }); // Robust fallback copy function for iframes function fallbackCopyText(text, callback) { const textArea = document.createElement('textarea'); textArea.value = text; // Prevent scrolling or zooming in mobile browsers textArea.style.position = 'fixed'; textArea.style.top = '0'; textArea.style.left = '0'; textArea.style.width = '2em'; textArea.style.height = '2em'; textArea.style.padding = '0'; textArea.style.border = 'none'; textArea.style.outline = 'none'; textArea.style.boxShadow = 'none'; textArea.style.background = 'transparent'; document.body.appendChild(textArea); textArea.focus(); textArea.select(); try { const successful = document.execCommand('copy'); if (successful && callback) { callback(); } } catch (err) { console.error('Fallback copy failed', err); } document.body.removeChild(textArea); }
